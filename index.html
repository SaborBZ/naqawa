import os
import glob
import shutil
import subprocess
import threading
import time
import re
import traceback
from fastapi import FastAPI, HTTPException
from fastapi.middleware.cors import CORSMiddleware
from fastapi.responses import FileResponse, JSONResponse
import uvicorn
import yt_dlp

app = FastAPI()

app.add_middleware(
    CORSMiddleware,
    allow_origins=["*"],
    allow_credentials=True,
    allow_methods=["*"],
    allow_headers=["*"],
)

CACHE_DIR = "/content/audio_cache"
os.makedirs(CACHE_DIR, exist_ok=True)

YTDL_BASE_OPTS = {
    'quiet': True,
    'no_warnings': True,
    'extractor_args': {
        'youtube': {
            'player_client': ['android', 'ios', 'mweb'],
            'player_skip': ['webpage', 'configs']
        }
    }
}

@app.get("/")
def health():
    return {"status": "online", "name": "Naqawa Engine", "gpu": "Tesla T4"}

@app.get("/process")
@app.post("/process")
async def process_video(url: str):
    print(f"\n🚀 [1/4] استلام الطلب: {url}")

    try:
        with yt_dlp.YoutubeDL(YTDL_BASE_OPTS) as ydl:
            info = ydl.extract_info(url, download=False)
            video_id = info.get("id")
    except Exception as e:
        print(f"❌ خطأ يوتيوب: {e}")
        return JSONResponse(status_code=400, content={"detail": f"خطأ في رابط اليوتيوب: {str(e)}"})

    cached_file = os.path.join(CACHE_DIR, f"{video_id}.mp3")

    if os.path.exists(cached_file):
        print(f"⚡ [الكاش] المقطع {video_id} جاهز مسبقاً!")
        return {"status": "ready", "video_id": video_id}

    work_dir = f"/content/work_{video_id}"
    os.makedirs(work_dir, exist_ok=True)

    try:
        print("⬇️ [2/4] جاري سحب ملف الصوت الخام...")
        raw_template = os.path.join(work_dir, "raw_audio.%(ext)s")
        download_opts = {
            **YTDL_BASE_OPTS,
            'format': 'bestaudio/best',
            'outtmpl': raw_template,
        }
        with yt_dlp.YoutubeDL(download_opts) as ydl:
            ydl.download([url])

        # العثور على الملف الذي تم تنزيله أياً كان امتداده
        downloaded = glob.glob(os.path.join(work_dir, "raw_audio.*"))
        if not downloaded:
            raise Exception("لم يتم العثور على ملف الصوت المحمل من يوتيوب")
        raw_file = downloaded[0]

        # تحويله إلى WAV معياري ليقرأه Demucs بأعلى سرعة
        input_wav = os.path.join(work_dir, "input.wav")
        subprocess.run(["ffmpeg", "-y", "-i", raw_file, "-vn", "-ar", "44100", "-ac", "2", input_wav], check=True)

        print("🧠 [3/4] تشغيل Demucs بالذكاء الاصطناعي على GPU...")
        cmd = ["demucs", "--two-stems", "vocals", "-d", "cuda", "-n", "htdemucs", "-o", work_dir, input_wav]
        res = subprocess.run(cmd, capture_output=True, text=True)
        if res.returncode != 0:
            print("Demucs Error:", res.stderr)
            raise Exception(f"Demucs failed: {res.stderr[:200]}")

        vocals_path = os.path.join(work_dir, "htdemucs", "input", "vocals.wav")
        if not os.path.exists(vocals_path):
            raise Exception("تعذر إيجاد ملف vocals.wav الناتج")

        print("🎛️ [4/4] تصدير وحفظ المسار الصوتي...")
        subprocess.run(["ffmpeg", "-y", "-i", vocals_path, "-b:a", "192k", cached_file], check=True)

        shutil.rmtree(work_dir, ignore_errors=True)
        print(f"✅ تمت معالجة الفيديو بنجاح: {video_id}")
        return {"status": "ready", "video_id": video_id}

    except Exception as e:
        traceback.print_exc()
        shutil.rmtree(work_dir, ignore_errors=True)
        print(f"❌ خطأ أثناء المعالجة: {e}")
        return JSONResponse(status_code=500, content={"detail": f"خطأ المعالجة: {str(e)}"})

@app.get("/audio/{video_id}")
async def stream_audio(video_id: str):
    file_path = os.path.join(CACHE_DIR, f"{video_id}.mp3")
    if os.path.exists(file_path):
        return FileResponse(file_path, media_type="audio/mpeg")
    raise HTTPException(status_code=404, detail="الملف الصوتي غير موجود")

def start_server():
    uvicorn.run(app, host="127.0.0.1", port=8000, log_level="warning")

threading.Thread(target=start_server, daemon=True).start()
time.sleep(2)

print("\n🌐 جاري فتح النفق العام...")
tunnel = subprocess.Popen(
    ["cloudflared", "tunnel", "--url", "http://127.0.0.1:8000"],
    stdout=subprocess.PIPE,
    stderr=subprocess.STDOUT,
    text=True
)

public_url = None
for line in tunnel.stdout:
    match = re.search(r'https://[a-zA-Z0-9-]+\.trycloudflare\.com', line)
    if match:
        public_url = match.group(0)
        break

print("\n" + "="*60)
print(f"🔥 انسخ هذا الرابط وضعه في نافذة إضافة نقاوة:\n{public_url}")
print("="*60 + "\n")

          🔍 1. ماذا يحدث في نظامك عند التثبيت؟ (Linux & Windows)
        


        

          

            
📦 عزل تام (Zero Pollution)

            
جميع مكتبات الذكاء الاصطناعي (PyTorch, Demucs) تُثبت داخل مجلد البيئة naqawa_env فقط، ولا يتم تعديل أو تنزيل أي شيء في بايثون النظام أو مسارات النظام العامة.

          

          

            
🧹 لا مساس بالسجل أو الإعدادات

            
على نظام Windows لا يتم تعديل الـ Registry أبداً، وعلى Linux لا يتم تعديل أي ملفات ضبط بالنظام (Config files). الأداة محمولة بالكامل (Portable).

          

          

            
🔒 خصوصية وأمان مطلق

            
في الخيار المحلي، تُعالج المقاطع وتُحفظ داخل مجلد audio_cache على قرصك، ولا يخرج أي مسار صوتي أو رابط خارج شبكتك الخاصة.
