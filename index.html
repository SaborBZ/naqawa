import os, glob, shutil, subprocess, threading, time, re
from fastapi import FastAPI, HTTPException
from fastapi.middleware.cors import CORSMiddleware
from fastapi.responses import FileResponse, JSONResponse
import uvicorn, yt_dlp

app = FastAPI()
app.add_middleware(CORSMiddleware, allow_origins=["*"], allow_credentials=True, allow_methods=["*"], allow_headers=["*"])

CACHE_DIR = "/content/audio_cache"
os.makedirs(CACHE_DIR, exist_ok=True)
YTDL_OPTS = {'quiet': True, 'extractor_args': {'youtube': {'player_client': ['android', 'ios', 'mweb'], 'player_skip': ['webpage', 'configs']}}}

@app.get("/process")
@app.post("/process")
async def process_video(url: str):
    with yt_dlp.YoutubeDL(YTDL_OPTS) as ydl:
        video_id = ydl.extract_info(url, download=False).get("id")
    cached_file = os.path.join(CACHE_DIR, f"{video_id}.mp3")
    if os.path.exists(cached_file): return {"status": "ready", "video_id": video_id}
    work_dir = f"/content/work_{video_id}"
    os.makedirs(work_dir, exist_ok=True)
    try:
        raw_tmpl = os.path.join(work_dir, "raw.%(ext)s")
        with yt_dlp.YoutubeDL({**YTDL_OPTS, 'format': 'bestaudio/best', 'outtmpl': raw_tmpl}) as ydl: ydl.download([url])
        raw_file = glob.glob(os.path.join(work_dir, "raw.*"))[0]
        input_wav = os.path.join(work_dir, "input.wav")
        subprocess.run(["ffmpeg", "-y", "-i", raw_file, "-vn", "-ar", "44100", "-ac", "2", input_wav], check=True)
        subprocess.run(["demucs", "--two-stems", "vocals", "-d", "cuda", "-n", "htdemucs", "-o", work_dir, input_wav], check=True)
        vocals = os.path.join(work_dir, "htdemucs", "input", "vocals.wav")
        subprocess.run(["ffmpeg", "-y", "-i", vocals, "-b:a", "192k", cached_file], check=True)
        shutil.rmtree(work_dir, ignore_errors=True)
        return {"status": "ready", "video_id": video_id}
    except Exception as e:
        shutil.rmtree(work_dir, ignore_errors=True)
        return JSONResponse(status_code=500, content={"detail": str(e)})

@app.get("/audio/{video_id}")
async def stream(video_id: str):
    file_path = os.path.join(CACHE_DIR, f"{video_id}.mp3")
    if os.path.exists(file_path): return FileResponse(file_path, media_type="audio/mpeg")
    raise HTTPException(status_code=404)

threading.Thread(target=lambda: uvicorn.run(app, host="127.0.0.1", port=8000, log_level="warning"), daemon=True).start()
time.sleep(2)
